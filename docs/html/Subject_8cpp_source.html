28void Subject::notify(const EventNotice &notice)
29{
30 // Non-obvious design decision: notify() iterates over a snapshot copy
31 // of the observers list rather than the member vector directly. An
32 // observer's own update() is free to attach() or detach() observers
33 // on this same Subject (CascadingZone::update() does exactly this
34 // indirectly, by cascading into further notify() calls elsewhere in
35 // the tree). If notify() walked the live `observers` vector while a
36 // callback mutated it, an iterator could be invalidated mid-loop.
37 // Copying first means every observer that was registered at the start
38 // of this notify() call is delivered to exactly once, and any
39 // attach()/detach() triggered by a callback only takes effect for the
40 // *next* notify() call.
41 std::vector<Observer *> snapshot = observers;
42 for (Observer *obs : snapshot)
43 {
44 if (obs)
45 {
46 obs->update(notice);
47 }
48 }
49}
